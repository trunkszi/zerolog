cmake_minimum_required(VERSION 4.0)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/ZerologToolchain.cmake")
zerolog_ensure_toolchain()

project(zerolog VERSION 0.1.0 DESCRIPTION "C++26 bounded low-latency logging" LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
if(NOT DEFINED BUILD_TESTING AND DEFINED ZEROLOG_BUILD_TESTS)
  set(BUILD_TESTING "${ZEROLOG_BUILD_TESTS}" CACHE BOOL "Build deterministic tests")
endif()
option(BUILD_TESTING "Build deterministic tests" ${PROJECT_IS_TOP_LEVEL})
option(ZEROLOG_BUILD_TESTS "Build deterministic tests" ${BUILD_TESTING})
option(ZEROLOG_BUILD_EXAMPLES "Build zerolog example binaries" ${PROJECT_IS_TOP_LEVEL})
option(ZEROLOG_BUILD_TOOLS "Build zerolog tools" ${PROJECT_IS_TOP_LEVEL})
option(ZEROLOG_INSTALL_PACKAGE "Install zerolog CMake package and headers" ${PROJECT_IS_TOP_LEVEL})
option(ZEROLOG_BUILD_BENCHES "Build benchmarks" OFF)
option(ZEROLOG_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
include(ZerologCompilerOptions)
include(ZerologReflection)
include(ZerologSanitizers)
include(ZerologTargets)
include(ZerologNegative)
include(ZerologCodegenGates)
zerolog_detect_reflection()
zerolog_create_libraries()

if(ZEROLOG_BUILD_TOOLS OR BUILD_TESTING)
  add_subdirectory(src)
endif()

if(ZEROLOG_BUILD_EXAMPLES OR BUILD_TESTING)
  add_subdirectory(examples)
endif()

if(BUILD_TESTING AND ZEROLOG_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
if(ZEROLOG_BUILD_BENCHES)
  add_subdirectory(benches)
endif()

if(ZEROLOG_INSTALL_PACKAGE)
  install(TARGETS zerolog_error zerolog_abi zerolog_sys zerolog_core zerolog_encode zerolog_format zerolog_sink zerolog EXPORT zerolog-targets)
  install(DIRECTORY include/ DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
  install(EXPORT zerolog-targets NAMESPACE zerolog:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/zerolog")
  configure_package_config_file(cmake/zerolog-config.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/zerolog-config.cmake"
    INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/zerolog")
  write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/zerolog-config-version.cmake" VERSION "${PROJECT_VERSION}" COMPATIBILITY SameMajorVersion)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/zerolog-config.cmake" "${CMAKE_CURRENT_BINARY_DIR}/zerolog-config-version.cmake" cmake/ZerologToolchain.cmake cmake/ZerologReflection.cmake cmake/ZerologCompilerOptions.cmake DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/zerolog")
endif()
